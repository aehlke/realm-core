cmake_minimum_required(VERSION 3.19)
project(RealmAdmissionPeerHarness LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(REALM_PEER_CORE_SOURCE "" CACHE PATH "Separate supplied pristine-old or staged-modern Core root")
set(REALM_PEER_ROLE "" CACHE STRING "old or modern; configure in separate build directories")
set(REALM_PEER_EXPECTED_DB_SHA256 "" CACHE STRING "Externally verified exact src/realm/db.cpp SHA256")
set(REALM_PEER_SOURCE_REVISION "" CACHE STRING "Externally verified pinned revision plus candidate identity")
option(REALM_PEER_BUILD_TESTS "Build SDK suites in the modern peer Core graph" OFF)
if(NOT REALM_PEER_ROLE STREQUAL "old" AND NOT REALM_PEER_ROLE STREQUAL "modern")
    message(FATAL_ERROR "REALM_PEER_ROLE must be old or modern")
endif()
if(NOT IS_DIRECTORY "${REALM_PEER_CORE_SOURCE}")
    message(FATAL_ERROR "Supply the intended separate Core source root")
endif()
file(REAL_PATH "${REALM_PEER_CORE_SOURCE}" _core)
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" _fixtures)
file(REAL_PATH "${CMAKE_CURRENT_BINARY_DIR}" _build)
string(FIND "${_build}/" "${_core}/" _inside_core)
string(FIND "${_build}/" "${_fixtures}/" _inside_fixtures)
if(_inside_core EQUAL 0 OR _inside_fixtures EQUAL 0)
    message(FATAL_ERROR "Use a separate output build directory")
endif()
if(DEFINED REALM_PEER_LOCKED_SOURCE AND NOT REALM_PEER_LOCKED_SOURCE STREQUAL _core)
    message(FATAL_ERROR "Do not reuse a peer build directory for another SDK root")
endif()
if(DEFINED REALM_PEER_LOCKED_ROLE AND NOT REALM_PEER_LOCKED_ROLE STREQUAL REALM_PEER_ROLE)
    message(FATAL_ERROR "Old and modern peers require separate build directories")
endif()
set(REALM_PEER_LOCKED_SOURCE "${_core}" CACHE INTERNAL "Fixed supplied Core source root")
set(REALM_PEER_LOCKED_ROLE "${REALM_PEER_ROLE}" CACHE INTERNAL "Fixed peer role")
string(LENGTH "${REALM_PEER_EXPECTED_DB_SHA256}" _hash_length)
if(NOT _hash_length EQUAL 64 OR NOT REALM_PEER_EXPECTED_DB_SHA256 MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "Supply the independently verified exact db.cpp SHA256")
endif()
if(NOT REALM_PEER_SOURCE_REVISION MATCHES "^[A-Za-z0-9._/+:-]+$")
    message(FATAL_ERROR "Supply an exact revision/candidate identifier without whitespace")
endif()
file(SHA256 "${_core}/src/realm/db.cpp" _db_sha)
string(TOLOWER "${REALM_PEER_EXPECTED_DB_SHA256}" _expected_db_sha)
if(NOT _db_sha STREQUAL _expected_db_sha)
    message(FATAL_ERROR "Supplied SDK db.cpp does not match expected candidate identity")
endif()
if(REALM_PEER_ROLE STREQUAL "old" AND EXISTS "${_core}/src/realm/session_admission.hpp")
    message(FATAL_ERROR "Old peer must use the pristine pinned SDK, not modern Core with a nil config")
endif()
if(REALM_PEER_ROLE STREQUAL "modern" AND NOT EXISTS "${_core}/src/realm/session_admission.hpp")
    message(FATAL_ERROR "Modern peer requires the staged admission implementation")
endif()
if(REALM_PEER_BUILD_TESTS AND NOT REALM_PEER_ROLE STREQUAL "modern")
    message(FATAL_ERROR "Only the modern peer may enable SDK tests")
endif()
if(REALM_PEER_BUILD_TESTS)
    # Core normally returns before defining tests when used as a subdirectory.
    # Keep one Storage/ObjectStore graph for the suites and the modern peer.
    enable_testing()
    set(REALM_CORE_SUBMODULE_BUILD OFF)
    set(REALM_NO_TESTS OFF CACHE BOOL "Enable SDK tests in the modern peer graph" FORCE)
    set(REALM_BUILD_LIB_ONLY OFF CACHE BOOL "Build SDK suites with the modern peer" FORCE)
else()
    set(REALM_CORE_SUBMODULE_BUILD ON)
    set(REALM_NO_TESTS ON CACHE BOOL "No SDK test runner in standalone peer build" FORCE)
    set(REALM_BUILD_LIB_ONLY ON CACHE BOOL "Build libraries only" FORCE)
endif()
add_subdirectory("${_core}" "${CMAKE_CURRENT_BINARY_DIR}/realm-core")
if(REALM_PEER_BUILD_TESTS AND (NOT TARGET CoreTests OR NOT TARGET ObjectStoreTests))
    message(FATAL_ERROR "The supplied modern Core did not expose both SDK suite targets")
endif()
set(_peer "realm_admission_${REALM_PEER_ROLE}")
add_executable(${_peer} peer.cpp)
target_link_libraries(${_peer} PRIVATE Realm::ObjectStore)
target_compile_definitions(${_peer} PRIVATE
    "REALM_PEER_ROLE=\"${REALM_PEER_ROLE}\""
    "REALM_PEER_DB_SHA256=\"${_db_sha}\"")
if(REALM_PEER_ROLE STREQUAL "modern")
    target_compile_definitions(${_peer} PRIVATE REALM_ADMISSION_PEER_MODERN=1)
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/peer.cpp" _peer_sha)
set(_provenance "role=${REALM_PEER_ROLE}\nsdk_tests=${REALM_PEER_BUILD_TESTS}\nsource_root=${_core}\nsource_revision=${REALM_PEER_SOURCE_REVISION}\ndb_sha256=${_db_sha}\nexpected_db_sha256=${_expected_db_sha}\npeer_sha256=${_peer_sha}\n")
foreach(_relative IN ITEMS src/realm/db.cpp src/realm/db.hpp src/realm/db_options.hpp
    src/realm/group.cpp src/realm/object-store/shared_realm.cpp src/realm/object-store/shared_realm.hpp
    src/realm/object-store/impl/realm_coordinator.cpp src/realm/session_admission.cpp src/realm/session_admission.hpp)
    if(EXISTS "${_core}/${_relative}")
        file(SHA256 "${_core}/${_relative}" _sha)
        string(APPEND _provenance "source_sha256.${_relative}=${_sha}\n")
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/provenance-$<CONFIG>.txt" CONTENT
    "${_provenance}executable=$<TARGET_FILE:${_peer}>\nstorage_library=$<TARGET_FILE:Storage>\nobjectstore_library=$<TARGET_FILE:ObjectStore>\n")
